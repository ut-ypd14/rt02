/**
 * 建立排程
 * Version: trigger_v0.1.3
 */

function setTrig() {
  clrTrig_();

  // 每班下班前
  dayTrig_('mailAuto_', 7, 45);
  dayTrig_('mailAuto_', 15, 50);
  dayTrig_('mailAuto_', 23, 45);

  // 每 5 分鐘重建 LINE VIEW
  minTrig_('laPing', 5);

  // 每月 2 號寄上個月資料
  monTrig_('mailMonth_', 2, 4, 10);

  // 每月 3 號定時清理表數據
  monTrig_('cleanData_', 3, 4, 10);

  return 'triggers created';
}

function clrTrig_() {
  const names = {
    laPing: true,
    mailAuto_: true,
    mailMonth_: true,
    cleanData_: true
  };

  const triggers = ScriptApp.getProjectTriggers();

  for (let i = 0; i < triggers.length; i++) {
    const fn = triggers[i].getHandlerFunction();

    if (names[fn]) {
      ScriptApp.deleteTrigger(triggers[i]);
    }
  }
}

function dayTrig_(fnName, hour, minute) {
  ScriptApp.newTrigger(fnName)
    .timeBased()
    .inTimezone(APP_TZ)
    .everyDays(1)
    .atHour(hour)
    .nearMinute(minute)
    .create();
}

function monTrig_(fnName, day, hour, minute) {
  ScriptApp.newTrigger(fnName)
    .timeBased()
    .inTimezone(APP_TZ)
    .onMonthDay(day)
    .atHour(hour)
    .nearMinute(minute)
    .create();
}

function minTrig_(fnName, minutes) {
  ScriptApp.newTrigger(fnName)
    .timeBased()
    .everyMinutes(minutes)
    .create();
}

function hourTrig_(fnName, hours) {
  ScriptApp.newTrigger(fnName)
    .timeBased()
    .everyHours(hours)
    .create();
}
